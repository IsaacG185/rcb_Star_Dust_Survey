def suggest_name_at(ra, dec):
    """Look up SIMBAD/Gaia at a sky position, return a best-guess label.
    Used only for the single-point paradigm-labeling lookup (one click =
    one query is fine here), not for bulk scanning."""
    coord = SkyCoord(ra * u.deg, dec * u.deg)
    try:
        simbad = Simbad()
        simbad.TIMEOUT = 30
        result = simbad.query_region(coord, radius=PARADIGM_MATCH_SEP_ARCSEC * u.arcsec)
        if result is not None and len(result) > 0:
            name = str(result[0]["MAIN_ID"])
            if not name.startswith("Gaia"):
                return name, "SIMBAD"
    except Exception as e:
        print("[SIMBAD lookup]", e)

    try:
        gaia_cat = get_plate_catalog_gaia_point(ra, dec, radius_arcsec=PARADIGM_MATCH_SEP_ARCSEC)
        if gaia_cat is not None and len(gaia_cat) > 0:
            return f"Gaia {gaia_cat[0]['source_id']}", "Gaia"
    except Exception as e:
        print("[Gaia lookup]", e)

    return "Unknown", "none"



# ============================================================
# Field-wide catalog queries (APASS + Gaia via VizieR ASU-TSV, ported from
# test_Simple_Photometry_Clayton.ipynb), plus SIMBAD. All cached by
# rounded field-center (ra, dec) so repeated/overlapping plate pointings
# reuse results instead of re-querying.
# ============================================================

def compute_field_radius_arcsec(wcs, shape, padding=1.05):
    """Radius (arcsec) of a cone search guaranteed to cover the WHOLE
    frame, computed from the actual image corners (not a fixed guess) --
    the separation from the field center to its farthest corner, with a
    5% padding margin."""
    ny, nx = shape
    center_ra, center_dec = wcs.all_pix2world(nx / 2.0, ny / 2.0, 0)
    corner_x = [0, nx - 1, 0, nx - 1]
    corner_y = [0, 0, ny - 1, ny - 1]
    corner_ra, corner_dec = wcs.all_pix2world(corner_x, corner_y, 0)
    center_coord = SkyCoord(ra=float(center_ra) * u.deg, dec=float(center_dec) * u.deg)
    corner_coords = SkyCoord(ra=corner_ra * u.deg, dec=corner_dec * u.deg)
    radius = center_coord.separation(corner_coords).max().to(u.arcsec).value * padding
    return float(center_ra), float(center_dec), radius

def _vizier_tsv_query(source, ra, dec, radius_arcsec, out_columns, max_rows=2000, timeout=30):
    """Raw VizieR ASU-TSV cone-search query -- same mechanism as
    test_Simple_Photometry_Clayton.ipynb's query_apass_field/
    query_gaia_field, generalized to any catalog/column set."""
    params = {
        "-source": source,
        "-c": f"{ra} {dec}",
        "-c.rs": radius_arcsec,
        "-out": out_columns,
        "-out.max": max_rows,
    }
    url = "https://vizier.cds.unistra.fr/viz-bin/asu-tsv?" + urllib.parse.urlencode(params)
    req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
    with urllib.request.urlopen(req, timeout=timeout) as response:
        return response.read().decode("utf-8")

def _parse_tsv_rows(raw_text, n_cols, converters):
    """Generic TSV row parser -- skips comments/headers, applies
    `converters` (a list of callables, one per column) to each field,
    silently dropping any row that fails to parse cleanly."""
    rows = []
    for line in raw_text.splitlines():
        line = line.strip()
        if not line or line.startswith("#") or line.startswith("-"):
            continue
        parts = line.split("\t")
        if len(parts) < n_cols:
            continue
        try:
            row = tuple(conv(p) for conv, p in zip(converters, parts[:n_cols]))
        except ValueError:
            continue
        rows.append(row)
    # Drop the header row if it slipped through (non-numeric first field).
    if rows and not isinstance(rows[0][0], (int, float)):
        rows = rows[1:]
    return rows

def _float_or_nan(s):
    s = s.strip()
    return float(s) if s else np.nan

def get_plate_catalog_apass(wcs, shape):
    """Every APASS DR9 star in the FULL frame (not just a fixed-radius
    guess), with a measured B magnitude. This is now the PRIMARY source
    list -- every returned star becomes a photometry target, replacing
    DAOStarFinder as the main detector (DAOStarFinder is kept only as a
    display fallback for plates where this yields ~nothing -- see
    detect_sources_fallback() below)."""
    ra_c, dec_c, radius = compute_field_radius_arcsec(wcs, shape)
    key = (round(ra_c, 4), round(dec_c, 4))
    if key in apass_field_cache:
        return apass_field_cache[key]

    try:
        raw = _vizier_tsv_query(
            "II/336/apass9", ra_c, dec_c, radius,
            "RAJ2000,DEJ2000,Bmag,e_Bmag"
        )
        rows = _parse_tsv_rows(raw, 3, [float, float, _float_or_nan])
        rows = [r for r in rows if not np.isnan(r[2])]
        apass_field_cache[key] = rows
        with open(APASS_FIELD_CACHE_FILE, "wb") as f:
            pickle.dump(apass_field_cache, f)
        print(f"[APASS] loaded {len(rows)} field stars (radius {radius:.0f}\")")
        return rows
    except Exception as e:
        print("[APASS ERROR]", e)
        apass_field_cache[key] = []
        return []

def get_plate_catalog_gaia(wcs, shape):
    """Every Gaia DR3 star in the full frame -- source_id (for "Gaia <id>"
    labeling) AND parallax/parallax_error (for distance/absolute
    magnitude) in one query."""
    ra_c, dec_c, radius = compute_field_radius_arcsec(wcs, shape)
    key = (round(ra_c, 4), round(dec_c, 4))
    if key in gaia_field_cache:
        return gaia_field_cache[key]

    try:
        raw = _vizier_tsv_query(
            "I/355/gaiadr3", ra_c, dec_c, radius,
            "RA_ICRS,DE_ICRS,Source,Plx,e_Plx"
        )
        rows = _parse_tsv_rows(raw, 5, [float, float, int, _float_or_nan, _float_or_nan])
        gaia_field_cache[key] = rows
        with open(GAIA_FIELD_CACHE_FILE, "wb") as f:
            pickle.dump(gaia_field_cache, f)
        print(f"[GAIA] loaded {len(rows)} field stars (radius {radius:.0f}\")")
        return rows
    except Exception as e:
        print("[GAIA ERROR]", e)
        gaia_field_cache[key] = []
        return []

def get_plate_catalog_gaia_point(ra, dec, radius_arcsec=3.0):
    """Small single-point Gaia lookup, used only by suggest_name_at() for
    the interactive paradigm-labeling flow (one click = one query)."""
    try:
        raw = _vizier_tsv_query("I/355/gaiadr3", ra, dec, radius_arcsec, "Source", max_rows=1)
        rows = _parse_tsv_rows(raw, 1, [int])
        return [{"source_id": r[0]} for r in rows]
    except Exception as e:
        print("[GAIA POINT ERROR]", e)
        return []

def get_plate_catalog_simbad(wcs, shape):
    """Every SIMBAD-catalogued object in the full frame, ONE cone search
    per plate pointing (not one query per star)."""
    ra_c, dec_c, radius = compute_field_radius_arcsec(wcs, shape)
    key = (round(ra_c, 4), round(dec_c, 4))
    if key in simbad_cache:
        return simbad_cache[key]

    try:
        simbad = Simbad()
        simbad.TIMEOUT = 60
        simbad.add_votable_fields("main_id", "ra", "dec")
        center = SkyCoord(ra_c * u.deg, dec_c * u.deg)
        result = simbad.query_region(center, radius=radius * u.arcsec)
        if result is None:
            result = []
        simbad_cache[key] = result
        with open(SIMBAD_CACHE_FILE, "wb") as f:
            pickle.dump(simbad_cache, f)
        print(f"[SIMBAD] loaded {len(result)} field objects")
        return result
    except Exception as e:
        print("[SIMBAD ERROR]", e)
        simbad_cache[key] = []
        return []

def match_detected_sources_simbad(ra, dec, catalog, max_sep_arcsec=5):
    if catalog is None or len(catalog) == 0:
        return ["Unknown"] * len(ra)
    try:
        cat_coords = SkyCoord(catalog["RA"], catalog["DEC"], unit=(u.hourangle, u.deg))
    except Exception as e:
        print("[SIMBAD MATCH]", e)
        return ["Unknown"] * len(ra)
    src_coords = SkyCoord(ra * u.deg, dec * u.deg)
    idx, sep, _ = match_coordinates_sky(src_coords, cat_coords)
    labels = []
    for j in range(len(src_coords)):
        if sep[j].arcsec <= max_sep_arcsec:
            name = str(catalog["MAIN_ID"][idx[j]])
            labels.append("Unknown" if name.startswith("Gaia") else name)
        else:
            labels.append("Unknown")
    return labels

def match_gaia_ids(ra, dec, gaia_rows, max_sep_arcsec=2.5):
    """Assigns 'Gaia <source_id>' to each position with a Gaia counterpart
    within max_sep_arcsec, else 'Unknown'."""
    n = len(ra)
    if not gaia_rows:
        return ["Unknown"] * n
    gaia_ra = np.array([r[0] for r in gaia_rows])
    gaia_dec = np.array([r[1] for r in gaia_rows])
    gaia_id = np.array([r[2] for r in gaia_rows])
    cat_coords = SkyCoord(ra=gaia_ra * u.deg, dec=gaia_dec * u.deg)
    src_coords = SkyCoord(ra=np.asarray(ra) * u.deg, dec=np.asarray(dec) * u.deg)
    idx, sep, _ = match_coordinates_sky(src_coords, cat_coords)
    labels = []
    for j in range(n):
        labels.append(f"Gaia {gaia_id[idx[j]]}" if sep[j].arcsec <= max_sep_arcsec else "Unknown")
    return labels

def update_name_cache(gaia_labels, simbad_names):
    changed = False
    for g, s in zip(gaia_labels, simbad_names):
        if s == "Unknown" or not g.startswith("Gaia "):
            continue
        try:
            gaia_name_cache[int(g.replace("Gaia ", ""))] = s
            changed = True
        except:
            pass
    if changed:
        with open(NAME_CACHE_FILE, "wb") as f:
            pickle.dump(gaia_name_cache, f)

def resolve_names(gaia_labels, simbad_names):
    labels = []
    for g, s in zip(gaia_labels, simbad_names):
        if s != "Unknown":
            labels.append(s)
            continue
        if g.startswith("Gaia "):
            try:
                sid = int(g.replace("Gaia ", ""))
                if sid in gaia_name_cache:
                    labels.append(gaia_name_cache[sid])
                    continue
            except:
                pass
        labels.append("Unknown")
    return labels

def resolve_gaia_label(label):
    if isinstance(label, str) and label.startswith("Gaia "):
        try:
            sid = int(label.replace("Gaia ", "").strip())
            if sid in gaia_name_cache:
                return gaia_name_cache[sid]
        except:
            pass
    return label

def match_against_paradigm(ra, dec, max_sep_arcsec=PARADIGM_MATCH_SEP_ARCSEC):
    all_entries = [e for entries in paradigm_db.values() for e in entries]
    if not all_entries:
        return ["Unknown"] * len(ra)
    cat_coords = SkyCoord([e["ra"] for e in all_entries], [e["dec"] for e in all_entries], unit="deg")
    src_coords = SkyCoord(np.asarray(ra) * u.deg, np.asarray(dec) * u.deg)
    idx, sep, _ = match_coordinates_sky(src_coords, cat_coords)
    labels = []
    for j in range(len(src_coords)):
        labels.append(all_entries[idx[j]]["label"] if sep[j].arcsec <= max_sep_arcsec else "Unknown")
    return labels

def get_plate_date(fits_path):
    try:
        header = astrofits.getheader(fits_path)
        for key in ["DATE-OBS", "DATE", "DATEOBS", "MJD-OBS"]:
            if key in header:
                return str(header[key])
        return "Unknown"
    except:
        return "Unknown"

def detect_plate_errors(data):

    errors = {
        "scratches":  [],
        "trailing":   [],
        "saturation": [],
        "dust":       [],
        "edge":       False,
        "dead_zone_fraction": 0.0,
        "saturation_area_fraction": 0.0,
    }

    h, w = data.shape

    lo, hi = np.nanpercentile(data, 1), np.nanpercentile(data, 99)
    if hi == lo:
        return errors
    norm = np.clip((data - lo) / (hi - lo), 0, 1)

    # Scratches
    try:
        scale   = 6
        small   = ndimage.zoom(norm, 1 / scale, order=1)
        sobel_h = ndimage.sobel(small, axis=0)
        sobel_v = ndimage.sobel(small, axis=1)
        edges   = np.hypot(sobel_h, sobel_v)
        edges   = (edges > np.percentile(edges, 97)).astype(np.uint8)

        tested_angles = np.linspace(-np.pi / 2, np.pi / 2, 90, endpoint=False)
        hspace, angles, dists = hough_line(edges, theta=tested_angles)

        peaks = hough_line_peaks(
            hspace, angles, dists,
            num_peaks=8,
            min_distance=20,
            threshold=0.35 * hspace.max()
        )

        hspace_mean = float(np.mean(hspace))
        hspace_std  = float(np.std(hspace))
        sig_thresh  = hspace_mean + 6 * hspace_std

        sh, sw = small.shape
        for peak_val, angle, dist in zip(*peaks):
            if peak_val < sig_thresh:
                continue

            cos_a, sin_a = np.cos(angle), np.sin(angle)
            if abs(sin_a) > 1e-6:
                x0_s, x1_s = 0, sw - 1
                y0_s = (dist - x0_s * cos_a) / sin_a
                y1_s = (dist - x1_s * cos_a) / sin_a
            else:
                y0_s, y1_s = 0, sh - 1
                x0_s = x1_s = dist / cos_a if abs(cos_a) > 1e-6 else 0

            y0_s = float(np.clip(y0_s, 0, sh - 1))
            y1_s = float(np.clip(y1_s, 0, sh - 1))
            x0_s = float(np.clip(x0_s, 0, sw - 1))
            x1_s = float(np.clip(x1_s, 0, sw - 1))

            errors["scratches"].append({
                "x0": x0_s * scale, "y0": y0_s * scale,
                "x1": x1_s * scale, "y1": y1_s * scale,
            })
    except Exception as e:
        print(f"[SCRATCH DETECT] {e}")

    # Trailing stars
    try:
        bright_mask = (norm > np.percentile(norm, 95)).astype(np.uint8)
        labeled, n_obj = ndimage.label(bright_mask)

        for obj_id in range(1, n_obj + 1):
            region = labeled == obj_id
            area   = region.sum()
            if area < 40 or area > 0.005 * h * w:
                continue

            coords = np.argwhere(region)
            if len(coords) < 10:
                continue

            cov  = np.cov(coords[:, 1], coords[:, 0])
            eigs = np.linalg.eigvalsh(cov)
            eigs = np.sort(np.abs(eigs))
            if eigs[0] < 1e-6:
                continue

            elongation = np.sqrt(eigs[1] / eigs[0])
            if elongation > 5.0:
                cy_r, cx_r = coords.mean(axis=0)
                angle = 0.5 * np.degrees(np.arctan2(
                    2 * cov[0, 1], cov[0, 0] - cov[1, 1]
                ))
                major = 2 * np.sqrt(eigs[1])
                minor = 2 * np.sqrt(eigs[0])
                errors["trailing"].append({
                    "x": float(cx_r), "y": float(cy_r),
                    "w": float(major), "h": float(minor),
                    "angle": float(angle)
                })
    except Exception as e:
        print(f"[TRAIL DETECT] {e}")

    # Saturation / blooming
    try:
        sat_thresh = np.percentile(norm, 99.9)
        sat_mask   = (norm >= sat_thresh).astype(np.uint8)
        sat_mask   = ndimage.binary_dilation(sat_mask, iterations=2).astype(np.uint8)

        errors["saturation_area_fraction"] = float(sat_mask.sum()) / float(h * w)

        labeled, n_obj = ndimage.label(sat_mask)

        for obj_id in range(1, n_obj + 1):
            region = labeled == obj_id
            area   = region.sum()
            if area < 200:
                continue

            coords  = np.argwhere(region)
            cy_r, cx_r = coords.mean(axis=0)
            ry = (coords[:, 0].max() - coords[:, 0].min()) / 2
            rx = (coords[:, 1].max() - coords[:, 1].min()) / 2
            r  = np.hypot(rx, ry)

            errors["saturation"].append({
                "x": float(cx_r), "y": float(cy_r), "r": float(r)
            })
    except Exception as e:
        print(f"[SAT DETECT] {e}")

    # Dust spots / emulsion streaks
    try:
        bg_size = max(15, min(h, w) // 12)
        local_bg = ndimage.uniform_filter(norm, size=bg_size)
        residual = norm - local_bg

        med_resid = float(np.nanmedian(residual))
        mad = float(np.nanmedian(np.abs(residual - med_resid)))
        sigma_est = max(1.4826 * mad, 1e-3)

        dark_mask = (residual <= (med_resid - 5 * sigma_est)).astype(np.uint8)
        dark_mask = ndimage.binary_opening(dark_mask, iterations=1).astype(np.uint8)
        labeled, n_obj = ndimage.label(dark_mask)

        for obj_id in range(1, n_obj + 1):
            region = labeled == obj_id
            area   = region.sum()
            if area < 15 or area > 0.02 * h * w:
                continue

            coords  = np.argwhere(region)
            cy_r, cx_r = coords.mean(axis=0)
            ry = (coords[:, 0].max() - coords[:, 0].min()) / 2
            rx = (coords[:, 1].max() - coords[:, 1].min()) / 2

            if rx < 1e-6 or ry < 1e-6:
                continue

            r = np.hypot(rx, ry)
            errors["dust"].append({
                "x": float(cx_r), "y": float(cy_r), "r": float(r)
            })
    except Exception as e:
        print(f"[DUST DETECT] {e}")

    # Dead zone / vignetting
    try:
        raw_min = np.nanmin(data)
        raw_max = np.nanmax(data)
        tol = max(1.0, 0.02 * (raw_max - raw_min))
        dead_mask = (data <= raw_min + tol).astype(np.uint8)
        labeled_dead, n_dead_obj = ndimage.label(dead_mask)
        largest_dead_area = 0
        if n_dead_obj > 0:
            sizes = ndimage.sum(dead_mask, labeled_dead, index=range(1, n_dead_obj + 1))
            if len(sizes):
                largest_dead_area = int(np.max(sizes))
        errors["dead_zone_fraction"] = float(largest_dead_area) / float(h * w)
    except Exception as e:
        print(f"[DEAD ZONE DETECT] {e}")
        errors["dead_zone_fraction"] = 0.0

    # Edge artifacts
    try:
        border = max(20, int(min(h, w) * 0.05))
        center_med = np.nanmedian(norm[border:h-border, border:w-border])
        center_std = np.nanstd(norm[border:h-border, border:w-border])

        edge_strips = [
            norm[:border, :],
            norm[h-border:, :],
            norm[:, :border],
            norm[:, w-border:],
        ]

        for strip in edge_strips:
            strip_med = np.nanmedian(strip)
            if abs(strip_med - center_med) > 3 * center_std:
                errors["edge"] = True
                break
    except Exception as e:
        print(f"[EDGE DETECT] {e}")

    return errors


def annotate_errors(ax, errors):

    legend_handles = []

    for s in errors["scratches"]:
        ax.plot(
            [s["x0"], s["x1"]], [s["y0"], s["y1"]],
            color="magenta", linewidth=1.2, alpha=0.8, linestyle="--"
        )
    if errors["scratches"]:
        legend_handles.append(
            Line2D([0], [0], color="magenta", linestyle="--", label=f'Scratch ×{len(errors["scratches"])}')
        )

    for t in errors["trailing"]:
        ellipse = mpatches.Ellipse(
            (t["x"], t["y"]),
            width=t["w"], height=t["h"],
            angle=t["angle"],
            edgecolor="orange", facecolor="none",
            linewidth=1.5, alpha=0.85
        )
        ax.add_patch(ellipse)
        ax.text(
            t["x"] + t["w"] / 2 + 3, t["y"],
            "trail", color="orange", fontsize=5,
            va="center"
        )
    if errors["trailing"]:
        legend_handles.append(
            mpatches.Patch(edgecolor="orange", facecolor="none",
                           label=f'Trailing ×{len(errors["trailing"])}')
        )

    for s in errors["saturation"]:
        circ = plt.Circle(
            (s["x"], s["y"]), s["r"],
            edgecolor="red", facecolor="none",
            linewidth=1.5, alpha=0.8, linestyle="-."
        )
        ax.add_patch(circ)
        ax.text(
            s["x"], s["y"] - s["r"] - 4,
            "sat", color="red", fontsize=5,
            ha="center"
        )
    if errors["saturation"]:
        legend_handles.append(
            mpatches.Patch(edgecolor="red", facecolor="none",
                           label=f'Saturation ×{len(errors["saturation"])}')
        )

    sat_frac = errors.get("saturation_area_fraction", 0.0)
    if sat_frac >= 0.05:
        legend_handles.append(
            mpatches.Patch(edgecolor="red", facecolor="red", alpha=0.3,
                           label=f"Sat area {sat_frac * 100:.0f}%")
        )

    for d in errors["dust"]:
        circ = plt.Circle(
            (d["x"], d["y"]), d["r"],
            edgecolor="deepskyblue", facecolor="none",
            linewidth=1.2, alpha=0.8, linestyle=":"
        )
        ax.add_patch(circ)
        ax.text(
            d["x"], d["y"] - d["r"] - 4,
            "dust", color="deepskyblue", fontsize=5,
            ha="center"
        )
    if errors["dust"]:
        legend_handles.append(
            mpatches.Patch(edgecolor="deepskyblue", facecolor="none",
                           label=f'Dust ×{len(errors["dust"])}')
        )

    dead_frac = errors.get("dead_zone_fraction", 0.0)
    if dead_frac >= 0.05:
        legend_handles.append(
            mpatches.Patch(edgecolor="gray", facecolor="gray", alpha=0.5,
                           label=f"Dead zone {dead_frac * 100:.0f}%")
        )

    if errors["edge"]:
        ax_h, ax_w = ax.get_ylim(), ax.get_xlim()
        img_h = abs(ax_h[1] - ax_h[0])
        img_w = abs(ax_w[1] - ax_w[0])
        rect = mpatches.Rectangle(
            (min(ax_w), min(ax_h)), img_w, img_h,
            edgecolor="lime", facecolor="none",
            linewidth=2.5, alpha=0.7
        )
        ax.add_patch(rect)
        legend_handles.append(
            mpatches.Patch(edgecolor="lime", facecolor="none",
                           label="Edge artifact")
        )

    if legend_handles:
        ax.legend(
            handles=legend_handles,
            loc="upper left",
            fontsize=6,
            framealpha=0.6,
            facecolor="black",
            labelcolor="white",
            edgecolor="gray"
        )

def detect_sources_fallback(fits_path):
    """DAOStarFinder (Cell 2's detect_sources()), used ONLY as a display
    aid when the APASS-based star list calibrates almost nothing (severe
    plate defects, or genuinely no APASS coverage). This fallback list is
    NEVER used for quality scoring -- only so the plate isn't empty when
    browsed, exactly as detection worked previously in this notebook."""
    try:
        sources, algorithm, data, data_sub, std, x_col, y_col = detect_sources(fits_path)
        if sources is None or len(sources) == 0:
            return None
        xs = np.array(sources[x_col], dtype=float)
        ys = np.array(sources[y_col], dtype=float)
        return {"x": xs, "y": ys, "algorithm": algorithm}
    except Exception as e:
        print(f"[FALLBACK DETECT] {fits_path.name}: {e}")
        return None


# ============================================================
# Quality classification. Hard defect gates are UNCHANGED from before.
# The "clean" signal is now APASS-match quality (how many/what fraction
# of the field's real APASS stars actually calibrate on this plate, and
# how tight the resulting fit is) instead of the old DAOStarFinder-
# detection-count heuristic -- plus, once a paradigm plate exists, a
# "matches the paradigm plate" signal.
# ============================================================
MIN_APASS_MATCHED_ABSOLUTE = 15
APASS_MATCH_FRACTION_THRESHOLD = 0.3
MAX_FIT_RMS = 0.5              # mag; a looser calibration isn't trustworthy even with plenty of matches
PARADIGM_RECOVERY_THRESHOLD = 0.5
PARADIGM_SLOPE_TOLERANCE = 0.35  # relative fractional tolerance on fit slope vs the paradigm plate's

def classify_plate_quality(errors, n_apass_total, apass_match_info=None,
                            lim_mag_apass=None, lim_mag_atlas=None, paradigm_match_info=None):

    n_scratches  = len(errors["scratches"])
    n_trailing   = len(errors["trailing"])
    n_saturation = len(errors["saturation"])
    n_dust       = len(errors["dust"])
    n_edge       = 1 if errors["edge"] else 0
    dead_zone_fraction = errors.get("dead_zone_fraction", 0.0)
    saturation_area_fraction = errors.get("saturation_area_fraction", 0.0)

    total_defects = n_scratches + n_trailing + n_saturation + n_dust + n_edge

    if dead_zone_fraction >= 0.08:
        return "defective"
    if saturation_area_fraction >= 0.05:
        return "defective"
    if (n_saturation >= 8) or (errors["edge"] and total_defects >= 6) or (n_apass_total < 3 and total_defects >= 3):
        return "defective"
    if total_defects >= 8:
        return "too_many_errors"

    defect_based_good = total_defects < 8

    apass_good = True
    if apass_match_info is not None:
        n_valid = apass_match_info.get("n_valid", 0)
        n_total = apass_match_info.get("n_total", 0)
        rms = apass_match_info.get("rms")
        frac = (n_valid / n_total) if n_total > 0 else 0.0
        match_ok = (frac >= APASS_MATCH_FRACTION_THRESHOLD) or (n_valid >= MIN_APASS_MATCHED_ABSOLUTE)
        fit_ok = (rms is None) or (rms <= MAX_FIT_RMS)
        apass_good = match_ok and fit_ok

    lim_mag = lim_mag_apass if lim_mag_apass is not None else lim_mag_atlas
    lim_mag_reference = _lim_mag_median_apass if lim_mag_apass is not None else _lim_mag_median_atlas
    lim_mag_is_deep = (lim_mag is None) or (lim_mag_reference is None) or (lim_mag >= lim_mag_reference)

    paradigm_good = True
    if paradigm_match_info is not None:
        paradigm_good = paradigm_match_info.get("consistent", True)

    if defect_based_good and apass_good and lim_mag_is_deep and paradigm_good:
        return "good_match"
    return "fair"


# ============================================================
# Per-plate initial-scan routine: defect detection, APASS field-wide star
# discovery, Gaia distances, full photometry (centroid refine, adaptive
# apertures, local background, saturation+FWHM), linearity fit, target
# star location+measurement, DAOStarFinder fallback if calibration is
# essentially empty, and quality classification (no paradigm yet).
# ============================================================

def process_plate_initial(f):
    hdr = astrofits.getheader(f)
    wcs = WCS(hdr)
    data = astrofits.getdata(f).astype(float)
    data = np.nan_to_num(data, nan=np.nanmedian(data))

    plate_errors = detect_plate_errors(data)

    apass_rows = get_plate_catalog_apass(wcs, data.shape)
    gaia_rows = get_plate_catalog_gaia(wcs, data.shape)

    ny, nx = data.shape
    edge_margin = DEFAULT_APERTURE_R + APERTURE_SAFETY_BUFFER

    if apass_rows:
        ra_arr = np.array([r[0] for r in apass_rows])
        dec_arr = np.array([r[1] for r in apass_rows])
        bmag_arr = np.array([r[2] for r in apass_rows])
        x_arr, y_arr = wcs.all_world2pix(ra_arr, dec_arr, 0)
        in_bounds = (
            (x_arr >= edge_margin) & (x_arr <= nx - 1 - edge_margin) &
            (y_arr >= edge_margin) & (y_arr <= ny - 1 - edge_margin)
        )
        star_ra = ra_arr[in_bounds]
        star_dec = dec_arr[in_bounds]
        star_bmag = bmag_arr[in_bounds]
        xs_init = x_arr[in_bounds]
        ys_init = y_arr[in_bounds]
    else:
        star_ra = np.array([]); star_dec = np.array([]); star_bmag = np.array([])
        xs_init = np.array([]); ys_init = np.array([])

    n_apass_total = len(xs_init)

    # Locate the target star via the 3-tier locate_target(): already an
    # APASS-list star (tier 1/1-widened), a direct WCS position that
    # centroid-refines onto a real local peak (tier 2), or a raw-pixel
    # local-threshold blob for badly overexposed/blooming epochs (tier 3).
    # Not found at all is a legitimate outcome too (V CrA is a variable
    # RCB star -- deep fades can put it below this plate's detection
    # limit entirely, which is real astrophysics, not a bug).
    target_idx_in_list = None
    target_tier = "not_found"
    if n_apass_total > 0:
        cat_coords = SkyCoord(ra=star_ra * u.deg, dec=star_dec * u.deg)
        sep = TARGET_STAR.separation(cat_coords)
        best_i = int(np.argmin(sep))
        if sep[best_i].arcsec < 10:
            target_idx_in_list = best_i
            target_tier = "apass_list"

    if target_idx_in_list is None:
        tx, ty, tier = locate_target(wcs, data, star_ra, star_dec, xs_init, ys_init)
        if tx is not None and tier not in ("apass_list", "apass_list_widened"):
            xs_init = np.append(xs_init, tx)
            ys_init = np.append(ys_init, ty)
            t_ra, t_dec = wcs.all_pix2world(tx, ty, 0)
            star_ra = np.append(star_ra, float(t_ra))
            star_dec = np.append(star_dec, float(t_dec))
            star_bmag = np.append(star_bmag, np.nan)
            target_idx_in_list = len(xs_init) - 1
            target_tier = tier
        elif tx is not None:
            # locate_target() itself found a widened APASS-list match
            # (target_idx not yet known here since that function returns
            # coordinates, not an index) -- find the matching index.
            d = np.hypot(xs_init - tx, ys_init - ty)
            target_idx_in_list = int(np.argmin(d))
            target_tier = tier

    n_total_positions = len(xs_init)

    if n_total_positions == 0:
        fallback = detect_sources_fallback(f)
        empty_match_info = {"n_valid": 0, "n_total": 0, "rms": None}
        quality = classify_plate_quality(plate_errors, 0, apass_match_info=empty_match_info)
        render = {
            "phase": "initial", "initial_scan_version": INITIAL_SCAN_VERSION,
            "errors": plate_errors,
            "star_ra": np.array([]), "star_dec": np.array([]),
            "star_x": np.array([]), "star_y": np.array([]), "star_bmag": np.array([]),
            "inst_mag": np.array([]), "final_flux": np.array([]), "aper_radii": np.array([]),
            "bkg_std": np.array([]), "is_saturated": np.array([], dtype=bool),
            "fwhm": np.array([]), "distance_pc": np.array([]), "abs_mag": np.array([]),
            "derived_b": np.array([]), "diff_ref_idx": None, "predicted_b": np.array([]),
            "gaia_names": [], "simbad_names": [], "resolved_names": [], "paradigm_names": [],
            "paradigm_version": paradigm_version,
            "calibration": None, "self_plate_limit": None,
            "target_idx": None, "target_found": False, "target_tier": "not_found",
            "target_x": None, "target_y": None,
            "target_inst_mag": np.nan, "target_calibrated_mag": np.nan,
            "target_calibrated_mag_diff": np.nan, "target_is_saturated": False,
            "fallback_sources": fallback, "apass_match_info": empty_match_info,
        }
        return {
            "tag": "empty", "quality": quality, "n": 0, "date": get_plate_date(f),
            "target": False, "human_verdict": None, "render": render,
        }

    xs_ref, ys_ref = refine_source_centroids(data, xs_init, ys_init)
    aper_radii = compute_adaptive_apertures(xs_ref, ys_ref)
    final_flux, raw_flux, bkg_std, apertures_list = local_background_photometry(data, xs_ref, ys_ref, aper_radii)
    is_saturated, max_vals = detect_source_saturation(data, xs_ref, ys_ref, apertures_list)
    fwhm_values = compute_wings_fwhm(data, xs_ref, ys_ref, is_saturated, max_vals)

    flux_for_mag = np.where(final_flux > 0, final_flux, np.nan)
    inst_mag = -2.5 * np.log10(flux_for_mag)

    star_bmag = np.asarray(star_bmag, dtype=float)

    # The target itself is excluded from the fit regardless of tier: even
    # if it came from the APASS list (tier 1), it's a variable star, so
    # its catalogued Bmag doesn't represent THIS epoch's brightness and
    # would be a bad calibration point.
    fit_exclude = is_saturated.copy()
    if target_idx_in_list is not None:
        fit_exclude[target_idx_in_list] = True
    calibration = compute_linearity_fit(inst_mag, star_bmag, fit_exclude)
    self_plate_limit = compute_plate_limit(bkg_std, calibration)

    distance_pc, abs_mag = compute_distances_and_absmag(star_ra, star_dec, star_bmag, gaia_rows)

    # Flux-ratio differential photometry -- a SECOND, independent
    # calibration cross-check for every star (including the target),
    # anchored to one bright unsaturated APASS-matched reference star
    # instead of the line fit's slope/intercept.
    derived_b, diff_ref_idx = compute_differential_photometry(final_flux, star_bmag, fit_exclude)

    # Distance-modulus prediction -- a THIRD, purely physical cross-check
    # (Gaia distance + absolute magnitude, no plate flux involved at all)
    # for every APASS-catalogued star. Not meaningful for the target
    # itself (it has no reliable absolute magnitude to predict FROM,
    # that's the unknown), so this is a QA diagnostic on the field stars.
    predicted_b = compute_predicted_b_distance(distance_pc, abs_mag, star_bmag, diff_ref_idx)

    ra_all, dec_all = wcs.pixel_to_world_values(xs_ref, ys_ref)
    ra_all = np.array(ra_all, dtype=float)
    dec_all = np.array(dec_all, dtype=float)

    gaia_names = match_gaia_ids(ra_all, dec_all, gaia_rows)
    simbad_catalog = get_plate_catalog_simbad(wcs, data.shape)
    simbad_names = match_detected_sources_simbad(ra_all, dec_all, simbad_catalog)
    update_name_cache(gaia_names, simbad_names)
    resolved_names = resolve_names(gaia_names, simbad_names)
    paradigm_names = match_against_paradigm(ra_all, dec_all)
    resolved_names = [p if p != "Unknown" else r for p, r in zip(paradigm_names, resolved_names)]
    resolved_names = [resolve_gaia_label(name) for name in resolved_names]

    target_found = False
    target_x = target_y = None
    target_inst_mag = np.nan
    target_calibrated_mag = np.nan
    target_calibrated_mag_diff = np.nan
    target_is_saturated = False
    if target_idx_in_list is not None and target_idx_in_list < len(xs_ref):
        ti = target_idx_in_list
        target_x, target_y = float(xs_ref[ti]), float(ys_ref[ti])
        target_is_saturated = bool(is_saturated[ti])
        if np.isfinite(final_flux[ti]) and final_flux[ti] > 0:
            target_found = True
            target_inst_mag = float(inst_mag[ti])
            if calibration is not None and calibration["slope"] != 0:
                target_calibrated_mag = (target_inst_mag - calibration["intercept"]) / calibration["slope"]
            if np.isfinite(derived_b[ti]):
                target_calibrated_mag_diff = float(derived_b[ti])

    n_valid_flux = int(np.sum(np.isfinite(final_flux) & (final_flux > 0) & np.isfinite(star_bmag)))
    apass_match_info = {
        "n_valid": n_valid_flux,
        "n_total": n_apass_total,
        "rms": calibration["rms"] if calibration is not None else None,
    }

    fallback = None
    if n_valid_flux < APASS_FALLBACK_MIN_STARS:
        fallback = detect_sources_fallback(f)

    lim_mag_apass, lim_mag_atlas = get_plate_limits(f)
    quality = classify_plate_quality(plate_errors, n_apass_total, apass_match_info, lim_mag_apass, lim_mag_atlas)

    if n_apass_total < 10:
        tag = "empty"
    elif n_apass_total < 100:
        tag = "crowded"
    else:
        tag = "rich"

    render = {
        "phase": "initial", "initial_scan_version": INITIAL_SCAN_VERSION,
        "errors": plate_errors,
        "star_ra": star_ra, "star_dec": star_dec,
        "star_x": xs_ref, "star_y": ys_ref, "star_bmag": star_bmag,
        "inst_mag": inst_mag, "final_flux": final_flux, "aper_radii": aper_radii,
        "bkg_std": bkg_std, "is_saturated": is_saturated, "fwhm": fwhm_values,
        "distance_pc": distance_pc, "abs_mag": abs_mag,
        "derived_b": derived_b, "diff_ref_idx": diff_ref_idx, "predicted_b": predicted_b,
        "gaia_names": gaia_names, "simbad_names": simbad_names,
        "resolved_names": resolved_names, "paradigm_names": paradigm_names,
        "paradigm_version": paradigm_version,
        "calibration": calibration, "self_plate_limit": self_plate_limit,
        "target_idx": target_idx_in_list, "target_tier": target_tier,
        "target_found": target_found, "target_x": target_x, "target_y": target_y,
        "target_inst_mag": target_inst_mag, "target_calibrated_mag": target_calibrated_mag,
        "target_calibrated_mag_diff": target_calibrated_mag_diff,
        "target_is_saturated": target_is_saturated,
        "fallback_sources": fallback, "apass_match_info": apass_match_info,
    }

    return {
        "tag": tag, "quality": quality, "n": n_apass_total, "date": get_plate_date(f),
        "target": target_found, "human_verdict": None, "render": render,
    }

def format_eta(seconds):
    if seconds is None or seconds != seconds or seconds < 0:
        return "calculating..."
    m, s = divmod(int(seconds), 60)
    h, m = divmod(m, 60)
    return f"{h:d}:{m:02d}:{s:02d}" if h else f"{m:d}:{s:02d}"

def render_scan_progress(n_done, total, start_time, status_word="Scanning", color="#4CAF50"):
    elapsed = time.monotonic() - start_time
    pct = (n_done / total * 100) if total else 0
    rate = (n_done / elapsed) if elapsed > 0 and n_done > 0 else 0
    remaining = ((total - n_done) / rate) if rate > 0 else None
    bar_width = 380
    filled = int(bar_width * pct / 100)
    scan_progress_html.value = f"""
    <div style="font-family: monospace; font-size: 13px; line-height: 1.5;">
      <div style="display:flex; align-items:center;">
        <div style="width:{bar_width}px; height:16px; background:#333;
                    border-radius:4px; overflow:hidden; margin-right:10px;">
          <div style="width:{filled}px; height:100%; background:{color};
                      transition: width 0.3s;"></div>
        </div>
        <b>{pct:5.1f}%</b>
      </div>
      <div style="margin-top:4px;">
        {status_word} &nbsp;
        <b>{n_done:,} / {total:,}</b> plates &nbsp;|&nbsp;
        <b>{rate:.2f}</b> plates/sec &nbsp;|&nbsp;
        Elapsed <b>{format_eta(elapsed)}</b> &nbsp;|&nbsp;
        ETA <b>{format_eta(remaining)}</b>
      </div>
    </div>
    """

def paradigm_established():

_initial_scan_has_run = False

def initial_scan(plates=None):
    """
    Phase 1: runs process_plate_initial() (APASS field discovery, Gaia
    distances, full photometry, defect detection, quality classification)
    on every plate. This is the FIRST scan the user sees -- it drives
    Review, the Paradigm picker, and the initial Filter categories, all
    based on real APASS-match quality rather than a DAOStarFinder-
    detection-count heuristic. Network-bound (APASS + Gaia + SIMBAD per
    plate, each cached), so this is intentionally single-threaded: the
    catalog-query functions above write to shared cache dicts AND pickle
    files, and doing that safely from multiple threads at once would need
    locking this build doesn't have yet.
    """
    target_plates = plates if plates is not None else cutouts
    total = max(len(target_plates), 1)
    scan_progress.max = total
    scan_progress.value = 0
    start_time = time.monotonic()
    _tick_state = {"last_render": 0.0}

    def _tick(n_done):
        scan_progress.value = n_done
        now = time.monotonic()
        if now - _tick_state["last_render"] > 0.15 or n_done == total:
            render_scan_progress(n_done, total, start_time, status_word="Initial scan", color="#4CAF50")
            _tick_state["last_render"] = now

    for i, f in enumerate(target_plates):
        cached = plate_db.get(str(f))
        already_done = (
            cached is not None and cached.get("render") is not None
            and cached["render"].get("initial_scan_version", -1) == INITIAL_SCAN_VERSION
        )
        if already_done:
            _tick(i + 1)
            continue

        scan_status.value = f.name
        try:
            entry = process_plate_initial(f)
            plate_db[str(f)] = entry
        except Exception as e:
            print(f"[INITIAL SCAN ERROR] {f.name}: {e}")
            plate_db[str(f)] = {
                "tag": "error", "quality": "defective", "n": 0,
                "date": get_plate_date(f), "target": False,
                "human_verdict": None, "render": None,
            }

        if (i + 1) % 25 == 0:
            with open(PLATE_DB_FILE, "wb") as fp:
                pickle.dump(plate_db, fp)
            print(f"Initial scan autosaved {i + 1} plates")

        _tick(i + 1)

    render_scan_progress(total, total, start_time, status_word="Initial scan complete", color="#4CAF50")
    scan_status.value = "Initial scan complete"
    with open(PLATE_DB_FILE, "wb") as fp:
        pickle.dump(plate_db, fp)
    print(f"Initial-scanned {len(plate_db)} plates")
    global _initial_scan_has_run
    _initial_scan_has_run = True
    refresh_filter_options()
    rebuild_dropdown()
    _refresh_review_queue()


def compute_paradigm_reference_slope():
    """Median calibration slope across every plate that currently has at
    least one paradigm label -- used as the "what a normal fit on this
    field looks like" reference for the paradigm-consistency check."""
    slopes = []
    for f_str in paradigm_db.keys():
        entry = plate_db.get(f_str)
        if entry is None or entry.get("render") is None:
            continue
        calib = entry["render"].get("calibration")
        if calib is not None:
            slopes.append(calib["slope"])
    return float(np.median(slopes)) if slopes else None


def paradigm_refinement_scan(plates=None):
    """
    Phase 2: runs AFTER a paradigm plate has been labeled. Entirely
    OFFLINE -- reuses every plate's already-cached star list, flux
    measurements, and calibration fit from initial_scan(); no network
    calls happen here. Re-resolves paradigm names against the (possibly
    just-updated) paradigm_db, computes a "matches the paradigm plate"
    signal (how many of the paradigm-labeled stars are recoverable on
    this plate, and how consistent this plate's own fit slope is with the
    reference slope across paradigm plates), and reclassifies quality
    with that folded in on top of the APASS-match signal.
    """
    global _initial_scan_has_run

    target_plates = plates if plates is not None else cutouts
    total = max(len(target_plates), 1)
    scan_progress.max = total
    scan_progress.value = 0
    start_time = time.monotonic()
    _tick_state = {"last_render": 0.0}

    def _tick(n_done):
        scan_progress.value = n_done
        now = time.monotonic()
        if now - _tick_state["last_render"] > 0.15 or n_done == total:
            render_scan_progress(n_done, total, start_time, status_word="Paradigm refinement", color="#2196F3")
            _tick_state["last_render"] = now

    all_paradigm_entries = [e for entries in paradigm_db.values() for e in entries]
    ref_slope = compute_paradigm_reference_slope()
    para_coords = None
    if all_paradigm_entries:
        para_coords = SkyCoord([e["ra"] for e in all_paradigm_entries], [e["dec"] for e in all_paradigm_entries], unit="deg")

    for i, f in enumerate(target_plates):
        cached = plate_db.get(str(f))
        if cached is None or cached.get("render") is None:
            _tick(i + 1)
            continue

        r = cached["render"]
        already_refined = (
            r.get("phase") == "refined"
            and r.get("paradigm_refinement_version", -1) == PARADIGM_REFINEMENT_VERSION
            and r.get("paradigm_version", -1) == paradigm_version
        )
        if already_refined:
            _tick(i + 1)
            continue

        try:
            star_ra = r.get("star_ra", np.array([]))
            star_dec = r.get("star_dec", np.array([]))

            paradigm_names = match_against_paradigm(star_ra, star_dec) if len(star_ra) > 0 else []
            r["paradigm_names"] = paradigm_names
            resolved = r.get("resolved_names", [])
            if resolved and paradigm_names:
                resolved = [p if p != "Unknown" else old for p, old in zip(paradigm_names, resolved)]
                resolved = [resolve_gaia_label(name) for name in resolved]
            r["resolved_names"] = resolved

            paradigm_match_info = None
            if all_paradigm_entries and len(star_ra) > 0:
                star_coords = SkyCoord(star_ra * u.deg, star_dec * u.deg)
                idx, sep, _ = match_coordinates_sky(para_coords, star_coords)
                final_flux = r.get("final_flux", np.array([]))
                recoverable = 0
                for k in range(len(all_paradigm_entries)):
                    if sep[k].arcsec <= PARADIGM_MATCH_SEP_ARCSEC:
                        si = idx[k]
                        if si < len(final_flux) and np.isfinite(final_flux[si]) and final_flux[si] > 0:
                            recoverable += 1
                recovery_fraction = recoverable / len(all_paradigm_entries)

                slope_consistent = True
                calib = r.get("calibration")
                if ref_slope is not None and calib is not None and ref_slope != 0:
                    rel_diff = abs(calib["slope"] - ref_slope) / abs(ref_slope)
                    slope_consistent = rel_diff <= PARADIGM_SLOPE_TOLERANCE

                consistent = (recovery_fraction >= PARADIGM_RECOVERY_THRESHOLD) and slope_consistent
                paradigm_match_info = {
                    "recovery_fraction": recovery_fraction,
                    "slope_consistent": slope_consistent,
                    "consistent": consistent,
                }
            elif all_paradigm_entries:
                paradigm_match_info = {"recovery_fraction": 0.0, "slope_consistent": True, "consistent": False}

            lim_mag_apass, lim_mag_atlas = get_plate_limits(f)
            quality = classify_plate_quality(
                r["errors"], plate_db[str(f)]["n"], r.get("apass_match_info"),
                lim_mag_apass, lim_mag_atlas, paradigm_match_info
            )

            r["paradigm_match_info"] = paradigm_match_info
            r["paradigm_version"] = paradigm_version
            r["phase"] = "refined"
            r["paradigm_refinement_version"] = PARADIGM_REFINEMENT_VERSION
            plate_db[str(f)]["quality"] = quality
            plate_db[str(f)]["render"] = r

        except Exception as e:
            print(f"[PARADIGM REFINEMENT ERROR] {f.name}: {e}")

        _tick(i + 1)

    render_scan_progress(total, total, start_time, status_word="Paradigm refinement complete", color="#2196F3")
    scan_status.value = "Paradigm refinement complete"
    with open(PLATE_DB_FILE, "wb") as fp:
        pickle.dump(plate_db, fp)
    print(f"Paradigm-refined {len(plate_db)} plates")
    _initial_scan_has_run = True
    refresh_filter_options()
    rebuild_dropdown()



# ============================================================
# Photometry engine (ported from test_Simple_Photometry_Clayton.ipynb):
# centroid refinement, crowding-aware adaptive apertures, local background
# via CircularAnnulus with neighbor-star exclusion, saturation detection,
# wings-only FWHM for saturated sources, a linearity fit (now with RMS),
# a self-derived plate-limit diagnostic, and two independent B-magnitude
# cross-checks (flux-ratio, and Gaia-distance-modulus).
# ============================================================

CENTROID_BOX = 21
DEFAULT_APERTURE_R = 40.0
MIN_APERTURE_R = 8.0
ANNULUS_WIDTH = 5.0
APERTURE_SAFETY_BUFFER = 2.0
SATURATION_FRACTION = 0.995
SATURATION_MIN_PIXELS = 5
PLATE_LIMIT_NSIGMA = 5
FWHM_EDGE_RADII = np.arange(25)
GAIA_MATCH_RADIUS_ARCSEC = 2.0
MIN_PARALLAX_SNR = 3.0
APASS_FALLBACK_MIN_STARS = 3  # below this many usable APASS stars, fall back to
                               # DAOStarFinder purely for display (see below)


def refine_source_centroids(data, xs, ys, box_size=CENTROID_BOX):
    xs = np.asarray(xs, dtype=float)
    ys = np.asarray(ys, dtype=float)
    if len(xs) == 0:
        return xs, ys
    try:
        xs_ref, ys_ref = centroid_sources(
            data, xs, ys, box_size=box_size, centroid_func=centroid_com
        )
        bad = ~np.isfinite(xs_ref) | ~np.isfinite(ys_ref) | (np.hypot(xs_ref - xs, ys_ref - ys) >= box_size)
        xs_ref = np.where(bad, xs, xs_ref)
        ys_ref = np.where(bad, ys, ys_ref)
        return xs_ref, ys_ref
    except Exception as e:
        print(f"[CENTROID REFINE] {e}")
        return xs, ys


def compute_adaptive_apertures(xs, ys, default_r=DEFAULT_APERTURE_R,
                                min_r=MIN_APERTURE_R, safety_buffer=APERTURE_SAFETY_BUFFER):
    n = len(xs)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([default_r])
    xs = np.asarray(xs, dtype=float)
    ys = np.asarray(ys, dtype=float)
    min_sep = np.full(n, np.inf)
    for i in range(n):
        d = np.hypot(xs - xs[i], ys - ys[i])
        d[i] = np.inf
        min_sep[i] = d.min()
    aper_radii = np.minimum(default_r, min_sep / 2 - safety_buffer)
    return np.clip(aper_radii, min_r, default_r)


def local_background_photometry(data, xs, ys, aper_radii, annulus_width=ANNULUS_WIDTH):
    n = len(xs)
    if n == 0:
        return np.array([]), np.array([]), np.array([]), []
    positions = list(zip(xs, ys))
    apertures = [CircularAperture(positions[i], r=float(aper_radii[i])) for i in range(n)]
    annuli = [
        CircularAnnulus(positions[i], r_in=float(aper_radii[i]), r_out=float(aper_radii[i]) + annulus_width)
        for i in range(n)
    ]

    combined_star_mask = np.zeros(data.shape, dtype=bool)
    for ap in apertures:
        m = ap.to_mask(method="center")
        sl_large, sl_small = m.get_overlap_slices(data.shape)
        if sl_large is None:
            continue
        combined_star_mask[sl_large] |= (m.data[sl_small] > 0)

    raw_sum = np.zeros(n)
    final_flux = np.zeros(n)
    bkg_std = np.full(n, np.nan)

    for i in range(n):
        t = aperture_photometry(data, apertures[i])
        raw_sum[i] = float(t["aperture_sum"][0])

        ann_mask_obj = annuli[i].to_mask(method="center")
        sl_large, sl_small = ann_mask_obj.get_overlap_slices(data.shape)

        bkg_median = 0.0
        if sl_large is not None:
            local_ring = ann_mask_obj.data[sl_small] > 0
            local_star = combined_star_mask[sl_large]
            local_data = data[sl_large]
            clean = local_ring & (~local_star)
            ann_pixels = local_data[clean]
            if ann_pixels.size >= 10:
                _, bkg_median, bkg_sigma = sigma_clipped_stats(ann_pixels)
                bkg_std[i] = bkg_sigma

        final_flux[i] = raw_sum[i] - bkg_median * apertures[i].area

    return final_flux, raw_sum, bkg_std, apertures


def detect_source_saturation(data, xs, ys, apertures,
                              sat_fraction=SATURATION_FRACTION, sat_min_pixels=SATURATION_MIN_PIXELS):
    n = len(xs)
    is_saturated = np.zeros(n, dtype=bool)
    max_vals = np.full(n, np.nan)
    for i in range(n):
        mask = apertures[i].to_mask(method="center")
        cut = mask.multiply(data)
        if cut is None:
            continue
        star_pixels = cut[mask.data > 0]
        if star_pixels.size == 0:
            continue
        max_val = float(star_pixels.max())
        max_vals[i] = max_val
        near_max_count = int(np.sum(star_pixels >= max_val * sat_fraction))
        is_saturated[i] = near_max_count > sat_min_pixels
    return is_saturated, max_vals


def compute_wings_fwhm(data, xs, ys, is_saturated, max_vals, sat_fraction=SATURATION_FRACTION,
                        edge_radii=FWHM_EDGE_RADII):
    n = len(xs)
    fwhm = np.full(n, np.nan)
    for i in range(n):
        if not is_saturated[i] or not np.isfinite(max_vals[i]):
            continue
        try:
            sat_thresh = max_vals[i] * sat_fraction
            sat_mask = data >= sat_thresh
            rp = RadialProfile(data, (float(xs[i]), float(ys[i])), edge_radii, mask=sat_mask)
            rp.gaussian_fit
            fwhm[i] = float(rp.gaussian_fwhm)
        except Exception as e:
            print(f"[WINGS FWHM] source at ({xs[i]:.1f},{ys[i]:.1f}): {e}")
    return fwhm


def compute_linearity_fit(inst_mag, catalog_mag, is_saturated):
    """Instrumental-vs-catalog linearity fit, saturated sources excluded.
    Returns {"slope","intercept","rms","n_used"} or None if too few points."""
    inst_mag = np.asarray(inst_mag)
    catalog_mag = np.asarray(catalog_mag)
    is_saturated = np.asarray(is_saturated)
    mask = np.isfinite(inst_mag) & np.isfinite(catalog_mag) & (~is_saturated)
    x = catalog_mag[mask]
    y = inst_mag[mask]
    if len(x) < 5:
        return None
    slope, intercept = np.polyfit(x, y, 1)
    resid = y - (slope * x + intercept)
    rms = float(np.sqrt(np.mean(resid ** 2)))
    return {"slope": float(slope), "intercept": float(intercept), "rms": rms, "n_used": int(len(x))}


def compute_plate_limit(bkg_std, calibration, n_sigma=PLATE_LIMIT_NSIGMA, reference_r=DEFAULT_APERTURE_R):
    valid_std = bkg_std[np.isfinite(bkg_std)]
    if valid_std.size == 0 or calibration is None:
        return None
    mean_bkg_std = float(np.mean(valid_std))
    reference_area = np.pi * reference_r ** 2
    threshold_flux = n_sigma * mean_bkg_std * np.sqrt(reference_area)
    if threshold_flux <= 0:
        return None
    threshold_inst_mag = -2.5 * np.log10(threshold_flux)
    slope = calibration["slope"]
    if slope == 0:
        return None
    return (threshold_inst_mag - calibration["intercept"]) / slope


def compute_differential_photometry(final_flux, catalog_b, is_saturated):
    """Flux-ratio ("measured") B-magnitude cross-check: anchors to the
    brightest unsaturated, APASS-matched source and transfers its real
    catalog B magnitude to every other source via flux ratio alone."""
    n = len(final_flux)
    derived_b = np.full(n, np.nan)
    candidates = [
        i for i in range(n)
        if np.isfinite(catalog_b[i]) and not is_saturated[i] and np.isfinite(final_flux[i]) and final_flux[i] > 0
    ]
    if not candidates:
        return derived_b, None
    ref = candidates[int(np.argmax(final_flux[candidates]))]
    ref_flux = final_flux[ref]
    for i in range(n):
        if i == ref:
            derived_b[i] = catalog_b[ref]
            continue
        if not (np.isfinite(final_flux[i]) and final_flux[i] > 0):
            continue
        derived_b[i] = catalog_b[ref] - 2.5 * np.log10(final_flux[i] / ref_flux)
    return derived_b, ref


def compute_distances_and_absmag(ra_arr, dec_arr, apass_b_mag_arr, gaia_rows,
                                  match_radius_arcsec=GAIA_MATCH_RADIUS_ARCSEC,
                                  min_parallax_snr=MIN_PARALLAX_SNR):
    """Gaia parallax cross-match -> distance (pc) and absolute B magnitude
    (distance modulus, no extinction correction) for every star with a
    good-enough parallax."""
    n = len(ra_arr)
    distance_pc = np.full(n, np.nan)
    abs_mag_B = np.full(n, np.nan)
    if not gaia_rows or n == 0:
        return distance_pc, abs_mag_B

    gaia_ra = np.array([r[0] for r in gaia_rows])
    gaia_dec = np.array([r[1] for r in gaia_rows])
    gaia_plx = np.array([r[3] for r in gaia_rows])
    gaia_eplx = np.array([r[4] for r in gaia_rows])

    star_coords = SkyCoord(ra=np.asarray(ra_arr) * u.deg, dec=np.asarray(dec_arr) * u.deg)
    gaia_coords = SkyCoord(ra=gaia_ra * u.deg, dec=gaia_dec * u.deg)
    idx, sep, _ = match_coordinates_sky(star_coords, gaia_coords)

    for i in range(n):
        if sep[i].arcsec > match_radius_arcsec:
            continue
        plx, eplx = gaia_plx[idx[i]], gaia_eplx[idx[i]]
        if np.isnan(plx) or plx <= 0:
            continue
        if np.isnan(eplx) or (plx / eplx) < min_parallax_snr:
            continue
        if np.isnan(apass_b_mag_arr[i]):
            continue
        d_pc = 1000.0 / plx
        distance_pc[i] = d_pc
        abs_mag_B[i] = apass_b_mag_arr[i] - 5 * np.log10(d_pc) + 5

    return distance_pc, abs_mag_B


def compute_predicted_b_distance(distance_pc, abs_mag_B, apass_b_mag, ref_idx):
    """Distance-modulus ('physical') B-magnitude cross-check: predicts
    each star's B magnitude from its OWN Gaia distance + absolute
    magnitude relative to the reference star's -- entirely independent of
    any flux measured on this plate."""
    n = len(distance_pc)
    predicted_b = np.full(n, np.nan)
    if ref_idx is None or np.isnan(distance_pc[ref_idx]) or np.isnan(abs_mag_B[ref_idx]):
        return predicted_b
    for i in range(n):
        if i == ref_idx:
            predicted_b[i] = apass_b_mag[ref_idx]
            continue
        if np.isnan(abs_mag_B[i]) or np.isnan(distance_pc[i]):
            continue
        delta = (abs_mag_B[i] - abs_mag_B[ref_idx]) + 5 * np.log10(distance_pc[i] / distance_pc[ref_idx])
        predicted_b[i] = apass_b_mag[ref_idx] + delta
    return predicted_b



# ============================================================
# Target-star location -- tiered, since V CrA is a variable star and
# won't always appear in APASS at its current brightness (or at all,
# during a deep RCB-type fading episode -- a real, expected non-detection,
# not a bug). Once located (any tier), the target is treated as just one
# more star appended to the batch, so it goes through the exact same
# centroid/aperture/background/saturation pipeline as everything else and
# gets calibrated via THIS plate's own linearity fit.
# ============================================================

def locate_target(wcs, data, star_ra, star_dec, star_x, star_y, primary_max_sep=10, widened_max_sep=15):
    """Returns (x, y, source_tier) for the target's best-guess pixel
    position, or (None, None, None) if it can't be located at all on this
    plate/epoch.

    Tier 1: already one of our APASS-derived stars (common case -- RCB
    stars are often bright enough to have an APASS entry).
    Tier 2: not in APASS, but its known WCS position is on-frame and
    centroid-refines cleanly (a real star, just not APASS-catalogued at
    this position/epoch).
    Tier 3: raw-pixel local-threshold blob fallback, for badly
    overexposed/blooming targets that never centroid cleanly."""
    if len(star_ra) > 0:
        cat_coords = SkyCoord(np.asarray(star_ra) * u.deg, np.asarray(star_dec) * u.deg)
        sep = TARGET_STAR.separation(cat_coords)
        idx = int(np.argmin(sep))
        if sep[idx].arcsec < primary_max_sep:
            return float(star_x[idx]), float(star_y[idx]), "apass_list"
        if sep[idx].arcsec < widened_max_sep:
            return float(star_x[idx]), float(star_y[idx]), "apass_list_widened"

    try:
        tx, ty = wcs.world_to_pixel_values(TARGET_STAR.ra.deg, TARGET_STAR.dec.deg)
        tx, ty = float(np.array(tx)), float(np.array(ty))
    except Exception as e:
        print(f"[TARGET LOCATE] {e}")
        return None, None, None

    margin = 5
    if not ((-margin <= tx < data.shape[1] + margin) and (-margin <= ty < data.shape[0] + margin)):
        return None, None, None

    # Tier 2: direct WCS position, centroid-refined.
    xr, yr = refine_source_centroids(data, np.array([tx]), np.array([ty]))
    if np.hypot(xr[0] - tx, yr[0] - ty) < CENTROID_BOX:
        # Only trust this if there's a real local peak, not blank sky --
        # a quick local-brightness sanity check.
        x0, x1 = max(int(xr[0] - 10), 0), min(int(xr[0] + 11), data.shape[1])
        y0, y1 = max(int(yr[0] - 10), 0), min(int(yr[0] + 11), data.shape[0])
        if x1 > x0 and y1 > y0:
            local_patch = data[y0:y1, x0:x1]
            if local_patch.max() > np.nanpercentile(data, 80):
                return float(xr[0]), float(yr[0]), "direct_position"

    # Tier 3: raw-pixel local-threshold blob (badly saturated/blooming target).
    try:
        search_r = 25
        x0 = max(int(tx - search_r), 0)
        x1 = min(int(tx + search_r) + 1, data.shape[1])
        y0 = max(int(ty - search_r), 0)
        y1 = min(int(ty + search_r) + 1, data.shape[0])
        if x1 > x0 and y1 > y0:
            local = data[y0:y1, x0:x1]
            local_thresh = np.nanpercentile(local, 90)
            bright_mask_local = local > local_thresh
            if bright_mask_local.any():
                ys_idx, xs_idx = np.nonzero(bright_mask_local)
                weights = local[ys_idx, xs_idx]
                cx_local = np.average(xs_idx, weights=weights)
                cy_local = np.average(ys_idx, weights=weights)
                return float(x0 + cx_local), float(y0 + cy_local), "raw_pixel_blob"
    except Exception as e:
        print(f"[TARGET FALLBACK] {e}")

    return None, None, None

    "ideal": "Ideal Image",
    "good_target": "Good Image, Target",

def categorize_plate(meta):
    quality = meta.get("quality", "fair")
    if meta.get("human_verdict") == "approved" and quality == "fair":
        quality = "good_match"
    target = bool(meta.get("target", False))
    is_good = quality in ("good_match", "fair")
    if quality == "good_match" and target:
        return "ideal"
    if target:
        return "good_target" if is_good else "defective_target"
    else:
        return "good_no_target" if is_good else "defective_no_target"

def filter_plates(cat_value, query, exclude_rejected=False):
    query = query.strip().lower()
    matches = []
    for f_str, meta in plate_db.items():
        f = Path(f_str)
        if exclude_rejected and meta.get("human_verdict") == "rejected":
            continue
        cat = categorize_plate(meta)
        if cat_value != "all" and cat != cat_value:
            continue
        if query and query not in f.name.lower():
            continue
        label = f"{f.name} | {CATEGORY_DISPLAY[cat]} | {meta['n']} APASS stars | {meta['date']}"
        q_rank = QUALITY_RANK.get(meta.get("quality", "fair"), 1)
        sort_key = (q_rank, -meta["n"])
        matches.append((label, f, sort_key))

    for f in cutouts:
        if str(f) not in plate_db:
            if query and query not in f.name.lower():
                continue
            if cat_value != "all":
                continue
            matches.append((f"{f.name} | not yet scanned", f, (99, 0)))

    matches.sort(key=lambda m: m[2])
    return [(label, f) for label, f, _ in matches]

def rebuild_paradigm_dropdown(*args):
    matches = filter_plates(category_filter.value, search_box.value, exclude_rejected=True)
    total_matches = len(matches)
    options = matches[:MAX_DROPDOWN_OPTIONS]
    paradigm_plate_dropdown.options = options
    if total_matches > MAX_DROPDOWN_OPTIONS:
        paradigm_dropdown_status.value = (
            f"Showing first {MAX_DROPDOWN_OPTIONS} of {total_matches} matches — "
            f"narrow with Search or the filter above to see the rest."
        )
    else:

# ============================================================
# Review queue -- deterministic signature-based grouping (not k-means),
# hard-capped at REVIEW_MAX_GROUPS, largest groups first. Same mechanism
# as before; the signature now also reflects APASS-match quality.
# ============================================================

REVIEW_MAX_GROUPS = 20

_fair_cluster_of = {}
_fair_cluster_members = {}

def _fair_plate_signature(meta):
    r = meta.get("render") or {}
    errs = r.get("errors", {})
    total_defects = (
        len(errs.get("scratches", [])) + len(errs.get("trailing", []))
        + len(errs.get("saturation", [])) + len(errs.get("dust", []))
        + (1 if errs.get("edge") else 0)
    )
    dead_bucket = min(int(errs.get("dead_zone_fraction", 0.0) * 20), 3)
    sat_bucket = min(int(errs.get("saturation_area_fraction", 0.0) * 20), 3)
    target_flag = bool(meta.get("target", False))
    n_bucket = min(int(meta.get("n", 0)) // 10, 10)
    apass_info = r.get("apass_match_info") or {}
    n_valid = apass_info.get("n_valid", 0)
    match_bucket = min(n_valid // 5, 6)
    return (min(total_defects, 8), dead_bucket, sat_bucket, target_flag, n_bucket, match_bucket)

def _cluster_fair_plates(max_groups=REVIEW_MAX_GROUPS):
    global _fair_cluster_of, _fair_cluster_members
    _fair_cluster_of = {}
    _fair_cluster_members = {}

    fair_items = [
        (f_str, meta) for f_str, meta in plate_db.items()
        if meta.get("quality") == "fair"
        and meta.get("human_verdict") is None
        and meta.get("render") is not None
    ]
    if not fair_items:
        return []

    groups = {}
    for f_str, meta in fair_items:
        sig = _fair_plate_signature(meta)
        groups.setdefault(sig, []).append(f_str)

    ranked = sorted(groups.items(), key=lambda kv: -len(kv[1]))[:max_groups]

    representatives = []
    for cluster_id, (sig, member_files) in enumerate(ranked):
        for mf in member_files:
            _fair_cluster_of[mf] = cluster_id
        _fair_cluster_members[cluster_id] = member_files
        member_files_sorted = sorted(member_files, key=lambda fs: plate_db[fs].get("n", 0))
        rep = member_files_sorted[len(member_files_sorted) // 2]
        representatives.append(rep)

    return representatives

review_plate_dropdown = widgets.Dropdown(description="Needs review:")
review_status = widgets.Label(value="")
approve_btn = widgets.Button(description="Approve (treat as good)", button_style="success")
reject_btn = widgets.Button(description="Reject (exclude from shortlist)", button_style="danger")
review_instructions = widgets.HTML(

    "plates sharing the same defect/APASS-match signature, largest groups "
    "first so each decision clears as many plates as possible. "
    "<b>Approve</b> treats this representative AND its whole group as good; "
    "<b>Reject</b> excludes them from the paradigm-plate shortlist below "
    "(they stay browsable in the main viewer once that's enabled).</small>"
)

def _pending_review_plates():
    representatives = _cluster_fair_plates()
    items = []
    for f_str in representatives:
        meta = plate_db.get(f_str)
        if meta is None:
            continue
        f = Path(f_str)
        r = meta["render"]
        errs = r.get("errors", {})
        apass_info = r.get("apass_match_info") or {}
        cluster_id = _fair_cluster_of.get(f_str)
        n_members = len(_fair_cluster_members.get(cluster_id, [f_str]))
        label = (
            f"{f.name} | represents {n_members} plate(s) | "
            f"APASS {apass_info.get('n_valid', 0)}/{apass_info.get('n_total', 0)} | "
            f"defects: scr{len(errs.get('scratches', []))} "
            f"trail{len(errs.get('trailing', []))} "
            f"sat{len(errs.get('saturation', []))} "
            f"dust{len(errs.get('dust', []))}"
        )
        items.append((n_members, label, f))
    items.sort(key=lambda t: -t[0])
    return [(label, f) for _, label, f in items]

def _refresh_review_queue():
    pending = _pending_review_plates()
    review_plate_dropdown.options = pending
    review_status.value = f"{len(pending)} representative plate(s) awaiting review."

def _approve_plate(_):
    fits_path = review_plate_dropdown.value
    if fits_path is None:
        return
    cluster_id = _fair_cluster_of.get(str(fits_path))
    members = _fair_cluster_members.get(cluster_id, [str(fits_path)])
    for f_str in members:
        if f_str in plate_db:
            plate_db[f_str]["human_verdict"] = "approved"
    with open(PLATE_DB_FILE, "wb") as fp:
        pickle.dump(plate_db, fp)
    print(f"Approved {len(members)} plate(s).")
    _refresh_review_queue()
    refresh_filter_options()
    rebuild_paradigm_dropdown()

def _reject_plate(_):
    fits_path = review_plate_dropdown.value
    if fits_path is None:
        return
    cluster_id = _fair_cluster_of.get(str(fits_path))
    members = _fair_cluster_members.get(cluster_id, [str(fits_path)])
    for f_str in members:
        if f_str in plate_db:
            plate_db[f_str]["human_verdict"] = "rejected"
    with open(PLATE_DB_FILE, "wb") as fp:
        pickle.dump(plate_db, fp)
    print(f"Rejected {len(members)} plate(s).")
    _refresh_review_queue()
    refresh_filter_options()
    rebuild_paradigm_dropdown()

approve_btn.on_click(_approve_plate)
reject_btn.on_click(_reject_plate)

def _teardown_shared_plot():
    fig = _viewer_state.get("fig")
    if fig is not None:
        for cid in _viewer_state.get("cids", []):
            try:
                fig.canvas.mpl_disconnect(cid)
            except Exception:
                pass
        plt.close(fig)
    _viewer_state.update({"fig": None, "ax": None, "cids": [], "highlight_marker": None,
                           "annotation": None, "star_scatter": None})

def _star_label_text(i):
    meta = _viewer_state["star_meta"][i]
    lines = [f"★ {TARGET_STAR_NAME}" if meta.get("is_target") else meta.get("name", "Unknown")]
    if np.isfinite(meta.get("bmag", np.nan)):
        lines.append(f"APASS B={meta['bmag']:.2f}")
    if np.isfinite(meta.get("inst_mag", np.nan)):
        lines.append(f"inst={meta['inst_mag']:.2f}")
    if np.isfinite(meta.get("calibrated_mag", np.nan)):
        lines.append(f"B={meta['calibrated_mag']:.2f} (fit)")
    if np.isfinite(meta.get("calibrated_mag_diff", np.nan)):
        lines.append(f"B={meta['calibrated_mag_diff']:.2f} (flux-ratio)")
    if np.isfinite(meta.get("distance_pc", np.nan)):
        lines.append(f"d={meta['distance_pc']:.0f}pc")
    if np.isfinite(meta.get("abs_mag", np.nan)):
        lines.append(f"M_B={meta['abs_mag']:.2f}")
    if meta.get("aper_r") is not None and meta["aper_r"] < DEFAULT_APERTURE_R:
        lines.append(f"r={meta['aper_r']:.0f}px")
    if meta.get("is_saturated"):
        fw = meta.get("fwhm")
        lines.append("SATURATED" + (f" FWHM={fw:.1f}" if fw is not None and np.isfinite(fw) else ""))
    if meta.get("paradigm_label"):
        lines.append(f"Paradigm: {meta['paradigm_label']}")
    return "\n".join(lines)

def _highlight_star(idx):
    ax = _viewer_state["ax"]
    if ax is None or idx is None or idx >= len(_viewer_state["xs"]):
        return
    x, y = _viewer_state["xs"][idx], _viewer_state["ys"][idx]
    hl = _viewer_state["highlight_marker"]
    hl.set_data([x], [y])
    hl.set_visible(True)
    ann = _viewer_state["annotation"]
    ann.xy = (x, y)
    ann.set_text(_star_label_text(idx))
    ann.set_visible(True)
    _viewer_state["fig"].canvas.draw_idle()
    with shared_info_output:
        clear_output(wait=True)
        print(_star_label_text(idx).replace("\n", "   |   "))

def _render_plate_view(fits_path, mode):
    """mode: 'readonly' (Review, Main viewer) or 'editable' (Paradigm labeling)."""
    _teardown_shared_plot()

    meta = plate_db.get(str(fits_path))
    if meta is None or meta.get("render") is None:
        with shared_plot_output:
            clear_output(wait=True)
            print("No data available for this plate.")
        with shared_info_output:
            clear_output(wait=True)
        return

    r = meta["render"]
    data = astrofits.getdata(fits_path)
    hdr = astrofits.getheader(fits_path)
    wcs = WCS(hdr)

    star_x = list(np.asarray(r.get("star_x", []), dtype=float))
    star_y = list(np.asarray(r.get("star_y", []), dtype=float))
    n_stars = len(star_x)

    star_bmag = np.asarray(r.get("star_bmag", np.full(n_stars, np.nan)))
    inst_mag = np.asarray(r.get("inst_mag", np.full(n_stars, np.nan)))
    calibration = r.get("calibration")
    derived_b = np.asarray(r.get("derived_b", np.full(n_stars, np.nan)))
    distance_pc = np.asarray(r.get("distance_pc", np.full(n_stars, np.nan)))
    abs_mag = np.asarray(r.get("abs_mag", np.full(n_stars, np.nan)))
    aper_radii = np.asarray(r.get("aper_radii", np.full(n_stars, DEFAULT_APERTURE_R)))
    is_saturated = np.asarray(r.get("is_saturated", np.zeros(n_stars, dtype=bool)))
    fwhm_vals = np.asarray(r.get("fwhm", np.full(n_stars, np.nan)))
    resolved_names = r.get("resolved_names", ["Unknown"] * n_stars)
    gaia_names = r.get("gaia_names", ["Unknown"] * n_stars)
    paradigm_names = r.get("paradigm_names", ["Unknown"] * n_stars)
    target_idx = r.get("target_idx")

    label_source = resolved_names if display_mode.value == "Object ID" else gaia_names

    calibrated_mag_arr = np.full(n_stars, np.nan)
    if calibration is not None and calibration["slope"] != 0:
        calibrated_mag_arr = (inst_mag - calibration["intercept"]) / calibration["slope"]

    star_meta = []
    manual_flags = [False] * n_stars
    for i in range(n_stars):
        star_meta.append({
            "name": label_source[i] if i < len(label_source) else "Unknown",
            "bmag": star_bmag[i] if i < len(star_bmag) else np.nan,
            "inst_mag": inst_mag[i] if i < len(inst_mag) else np.nan,
            "calibrated_mag": calibrated_mag_arr[i] if i < len(calibrated_mag_arr) else np.nan,
            "calibrated_mag_diff": derived_b[i] if i < len(derived_b) else np.nan,
            "distance_pc": distance_pc[i] if i < len(distance_pc) else np.nan,
            "abs_mag": abs_mag[i] if i < len(abs_mag) else np.nan,
            "aper_r": aper_radii[i] if i < len(aper_radii) else DEFAULT_APERTURE_R,
            "is_saturated": bool(is_saturated[i]) if i < len(is_saturated) else False,
            "fwhm": fwhm_vals[i] if i < len(fwhm_vals) else np.nan,
            "is_target": (target_idx is not None and i == target_idx),
            "paradigm_label": paradigm_names[i] if i < len(paradigm_names) and paradigm_names[i] != "Unknown" else None,
            "ra": r["star_ra"][i] if "star_ra" in r and i < len(r["star_ra"]) else None,
            "dec": r["star_dec"][i] if "star_dec" in r and i < len(r["star_dec"]) else None,
        })

    fallback = r.get("fallback_sources")
    fallback_used = n_stars == 0 and fallback is not None
    if fallback_used:
        star_x = list(np.asarray(fallback["x"], dtype=float))
        star_y = list(np.asarray(fallback["y"], dtype=float))
        n_stars = len(star_x)
        star_meta = [{"name": "Unknown (fallback detection)", "is_target": False} for _ in range(n_stars)]
        manual_flags = [False] * n_stars

    with shared_plot_output:
        clear_output(wait=True)
        fig, ax = plt.subplots(figsize=(9, 9))
        try:
            ax.imshow(data, origin="lower", cmap="gray", norm=LogNorm())
        except Exception:
            ax.imshow(data, origin="lower", cmap="gray")  # LogNorm can fail on all-zero/negative data
        title_extra = "  (DAOStarFinder fallback -- APASS calibrated ~nothing here)" if fallback_used else ""
        edit_hint = "  [left-click: label/add, right-click: remove]" if mode == "editable" else "  [click a star for details]"
        ax.set_title(f"{fits_path.name}{title_extra}{edit_hint}", fontsize=10)

        star_colors = ["orange" if m else "red" for m in manual_flags]
        star_scatter = ax.scatter(star_x, star_y, s=70, facecolors="none", edgecolors=star_colors,
                                   linewidths=1.3, picker=True, pickradius=8)

        sat_idx = [i for i in range(n_stars) if star_meta[i].get("is_saturated")]
        if sat_idx:
            ax.scatter([star_x[i] for i in sat_idx], [star_y[i] for i in sat_idx],
                       s=100, marker="x", color="magenta", linewidths=1.5)
        if target_idx is not None and target_idx < len(star_x) and not fallback_used:
            ax.scatter([star_x[target_idx]], [star_y[target_idx]], s=170,
                       edgecolors="cyan", facecolors="none", linewidths=2)

        highlight_marker, = ax.plot([], [], "o", markersize=24, markerfacecolor="none",
                                     markeredgecolor="#FFD700", markeredgewidth=3, zorder=5)
        annotation = ax.annotate("", xy=(0, 0), xytext=(15, 15), textcoords="offset points",
                                  color="black", fontsize=9,
                                  bbox=dict(boxstyle="round", fc="yellow", alpha=0.9),
                                  arrowprops=dict(arrowstyle="->"))
        annotation.set_visible(False)

        self_limit = r.get("self_plate_limit")
        if self_limit is not None:
            ax.text(0.02, 0.98, f"Plate limit (5σ): B ≈ {self_limit:.2f}",
                    transform=ax.transAxes, color="yellow", fontsize=10, va="top",
                    bbox=dict(facecolor="black", alpha=0.6, edgecolor="none"))

        if error_toggle.value == "Show Errors":
            annotate_errors(ax, r.get("errors", {}))

        prior_legend = ax.get_legend()
        extra_handles = []
        if sat_idx:
            extra_handles.append(Line2D([0], [0], marker="x", color="magenta", linestyle="None",
                                         markersize=8, label=f"Saturated ×{len(sat_idx)}"))
        if target_idx is not None and target_idx < len(star_x) and not fallback_used:
            extra_handles.append(Line2D([0], [0], marker="o", color="cyan", linestyle="None",
                                         markersize=10, markerfacecolor="none", label=TARGET_STAR_NAME))
        if extra_handles:
            ax.legend(handles=extra_handles, loc="upper right", fontsize=7, framealpha=0.6,
                      facecolor="black", labelcolor="white", edgecolor="gray")
            if prior_legend is not None:
                ax.add_artist(prior_legend)

        _viewer_state.update({
            "fig": fig, "ax": ax, "fits_path": fits_path, "mode": mode,
            "xs": star_x, "ys": star_y, "wcs": wcs, "data": data,
            "highlight_marker": highlight_marker, "annotation": annotation,
            "star_scatter": star_scatter, "star_meta": star_meta,
            "manual": manual_flags, "cids": [],
        })

        cids = []
        if mode == "readonly":
            def _on_pick(event):
                if event.artist != _viewer_state["star_scatter"]:
                    return
                _highlight_star(event.ind[0])
            cids.append(fig.canvas.mpl_connect("pick_event", _on_pick))
        else:
            cids.append(fig.canvas.mpl_connect("button_press_event", _on_paradigm_click))
        _viewer_state["cids"] = cids

        plt.show()

    with shared_info_output:
        clear_output(wait=True)
        if mode == "editable":
            print("Click a red/orange marker to label it, empty space to add one, "
                  "or right-click a marker to remove it.")
        elif n_stars == 0:
            print("No stars to show for this plate.")
        else:
            print(f"{n_stars} star(s) shown. Click one for details.")

def _label_point(x, y, ra, dec):
    label_type = widgets.ToggleButtons(options=["Object ID", "GAIA ID"], value="Object ID", description="Type:")
    label_box = widgets.Text(value="", description="Label:",
                              placeholder="looking up suggestion... (or just start typing)")
    confirm_btn = widgets.Button(description="Confirm", button_style="success")
    skip_btn = widgets.Button(description="Skip", button_style="")

    def _confirm(_):
        raw = label_box.value.strip()
        if label_type.value == "GAIA ID":
            digits = raw.replace("Gaia", "").strip()
            if digits.isdigit():
                final_label = f"Gaia {digits}"
            else:
                with paradigm_output:
                    clear_output(wait=True)
                    print(f"'{raw}' isn't a numeric Gaia source_id — enter digits only.")
                return
        else:
            final_label = raw

        fits_path = _viewer_state["fits_path"]
        paradigm_db.setdefault(str(fits_path), [])
        paradigm_db[str(fits_path)].append({"ra": ra, "dec": dec, "label": final_label, "source": "manual/suggested"})
        save_paradigm_db()
        bump_paradigm_version()

        with paradigm_output:
            clear_output(wait=True)
            print(f"Saved: {final_label}  (RA={ra:.5f}, Dec={dec:.5f})")
            print("Label more objects, then click 'Apply Paradigm' below when done.")
        _refresh_paradigm_table()

    def _skip(_):
        with paradigm_output:
            clear_output(wait=True)

    confirm_btn.on_click(_confirm)
    skip_btn.on_click(_skip)

    with paradigm_output:
        clear_output(wait=True)
        print(f"Point at pixel ({x:.1f}, {y:.1f})  ->  RA={ra:.5f}, Dec={dec:.5f}")
        print("Looking up a name suggestion in the background...")
        display(widgets.HBox([label_type, label_box, confirm_btn, skip_btn]))

    def _do_lookup():
        try:
            suggestion, src = suggest_name_at(ra, dec)
        except Exception as e:
            suggestion, src = "Unknown", f"error: {e}"
        if not label_box.value.strip():
            suggested_mode = "GAIA ID" if suggestion.startswith("Gaia ") else "Object ID"
            label_type.value = suggested_mode
            label_box.value = suggestion.replace("Gaia ", "") if suggested_mode == "GAIA ID" else suggestion
        with paradigm_output:
            print(f"Auto-suggested: '{suggestion}' (source: {src})")

    Thread(target=_do_lookup, daemon=True).start()

def _add_paradigm_marker(x, y):
    wcs = _viewer_state["wcs"]
    ra, dec = wcs.pixel_to_world_values(x, y)
    ra, dec = float(np.array(ra)), float(np.array(dec))
    _viewer_state["xs"].append(float(x))
    _viewer_state["ys"].append(float(y))
    _viewer_state["manual"].append(True)
    _viewer_state["star_meta"].append({"name": "manual", "is_target": False})
    sc = _viewer_state["star_scatter"]
    sc.set_offsets(np.column_stack([_viewer_state["xs"], _viewer_state["ys"]]))
    colors = ["orange" if m else "red" for m in _viewer_state["manual"]]
    sc.set_edgecolor(colors)
    _viewer_state["fig"].canvas.draw_idle()
    _label_point(x, y, ra, dec)

def _remove_paradigm_marker(j):
    wcs = _viewer_state["wcs"]
    x, y = _viewer_state["xs"][j], _viewer_state["ys"][j]
    ra, dec = wcs.pixel_to_world_values(x, y)
    ra, dec = float(np.array(ra)), float(np.array(dec))

    del _viewer_state["xs"][j]
    del _viewer_state["ys"][j]
    del _viewer_state["manual"][j]
    del _viewer_state["star_meta"][j]
    sc = _viewer_state["star_scatter"]
    if _viewer_state["xs"]:
        sc.set_offsets(np.column_stack([_viewer_state["xs"], _viewer_state["ys"]]))
        colors = ["orange" if m else "red" for m in _viewer_state["manual"]]
        sc.set_edgecolor(colors)
    else:
        sc.set_offsets(np.empty((0, 2)))
    _viewer_state["fig"].canvas.draw_idle()

    fits_path = _viewer_state["fits_path"]
    entries = paradigm_db.get(str(fits_path), [])
    entries = [e for e in entries if not (abs(e["ra"] - ra) < 1e-6 and abs(e["dec"] - dec) < 1e-6)]
    paradigm_db[str(fits_path)] = entries
    save_paradigm_db()
    bump_paradigm_version()

    with paradigm_output:
        clear_output(wait=True)
        print(f"Removed marker at pixel ({x:.1f}, {y:.1f}).")
    _refresh_paradigm_table()

def _on_paradigm_click(event):
    if event.inaxes != _viewer_state["ax"] or event.xdata is None:
        return
    xs, ys = _viewer_state["xs"], _viewer_state["ys"]

    if event.button == 3:
        if not xs:
            with paradigm_output:
                clear_output(wait=True)
                print("No marker close enough to remove.")
            return
        d = np.hypot(np.array(xs) - event.xdata, np.array(ys) - event.ydata)
        j = int(np.argmin(d))
        if d[j] > PARADIGM_CLICK_PX:
            with paradigm_output:
                clear_output(wait=True)
                print("No marker close enough to remove.")
            return
        _remove_paradigm_marker(j)
        return

    if xs:
        d = np.hypot(np.array(xs) - event.xdata, np.array(ys) - event.ydata)
        j = int(np.argmin(d))
        if d[j] <= PARADIGM_CLICK_PX:
            _highlight_star(j)
            wcs = _viewer_state["wcs"]
            ra, dec = wcs.pixel_to_world_values(xs[j], ys[j])
            ra, dec = float(np.array(ra)), float(np.array(dec))
            _label_point(xs[j], ys[j], ra, dec)
            return

    _add_paradigm_marker(event.xdata, event.ydata)

def _clear_plate(_):
    fits_path = _viewer_state.get("fits_path")
    if fits_path is None:
        with paradigm_output:
            clear_output(wait=True)
            print("No plate is currently open for labeling.")
        return
    n_labels = len(paradigm_db.get(str(fits_path), []))
    paradigm_db.pop(str(fits_path), None)
    save_paradigm_db()
    bump_paradigm_version()
    _render_plate_view(fits_path, "editable")
    with paradigm_output:
        clear_output(wait=True)
        print(f"Cleared plate: removed {n_labels} saved label(s).")
    _refresh_paradigm_table()

def _copy_name_to_search(_):
    fits_path = _viewer_state.get("fits_path")
    if fits_path is None:
        with paradigm_output:
            clear_output(wait=True)
            print("No plate is currently open for labeling.")
        return
    search_box.value = fits_path.name
    with paradigm_output:
        clear_output(wait=True)
        print(f"Copied '{fits_path.name}' into the Search box above.")

def _apply_paradigm(_):
    if not paradigm_established():
        paradigm_status.value = "Label at least one object on a paradigm plate before applying."
        return
    paradigm_status.value = "Running paradigm refinement across all plates..."
    paradigm_refinement_scan()
    paradigm_status.value = "Done — paradigm refinement complete, filters updated."
